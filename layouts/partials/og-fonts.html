{{- /* og-fonts.html - the Pretendard TrueType files the share card is drawn with (layouts/partials/og-card.html), as a dict of weight to resource. Call it with partialCached: the fonts are fetched once per build (Hugo keeps remote resources in its file cache) and checked against the SHA-256 jsDelivr publishes for this release, the same pin scripts/subset-fonts.py uses for the web fonts. The card draws whole titles, so it needs the full fonts, not the site's subset. */ -}} {{- $base := "https://cdn.jsdelivr.net/npm/pretendard@1.3.9/dist/public/static/alternative/" -}} {{- $pins := dict "Bold" "c16b88c670d23e83fa1170c954cbc4822d3b8dad3c3cde15d798a94b43d97985" "Medium" "3bae579377eb8e9ac412cb4809ebc3de1d956ed75995c1e346f0c1311053f4e2" -}} {{- $fonts := dict -}} {{- range $weight, $sum := $pins -}} {{- $font := resources.GetRemote (printf "%sPretendard-%s.ttf" $base $weight) -}} {{- if ne (crypto.SHA256 $font.Content) $sum -}} {{- errorf "og-fonts: Pretendard-%s.ttf does not match its pinned SHA-256" $weight -}} {{- end -}} {{- $fonts = merge $fonts (dict $weight $font) -}} {{- end -}} {{- return $fonts -}}
