{{- /* og-card.html - the 1200x630 share card for a page, returned as an image resource for og:image / twitter:image (head.html). Drawn at build time on assets/og/card-base.png, all of it centered in the middle 560px: the wordmark next to the menu as skeleton pills (the compact header, mid-scroll), then the first category, the title and the summary, then the domain and date under a rule. Feeds that crop link images to the center square (Facebook on phones) still show every part of it, and X, which shows no title or description under a large card, gets both in the image. Pages that aren't posts (home, lists, tags) get the site description as the title. */ -}} {{- $fonts := partialCached "og-fonts.html" . "og-fonts" -}} {{- $ink := "#1e1e1e" -}} {{- $muted := "#6c6c6c" -}} {{- $accent := "#1f6fd1" -}} {{- $heading := site.Params.description -}} {{- $summary := "" -}} {{- $eyebrow := "" -}} {{- $date := "" -}} {{- if .IsPage -}} {{- $heading = .Title -}} {{- $summary = .Description | default (.Summary | plainify | htmlUnescape) -}} {{- with .Params.categories }}{{ $eyebrow = index . 0 | upper }}{{ end -}} {{- if not .Date.IsZero }}{{ $date = .Date.Format "Jan 02, 2006" }}{{ end -}} {{- end -}} {{- $heading = replaceRE `\s+` " " $heading | strings.TrimSpace -}} {{- $summary = replaceRE `\s+` " " $summary | strings.TrimSpace -}} {{- $footer := (urls.Parse site.BaseURL).Host -}} {{- with $date }}{{ $footer = printf "%s / %s" $footer . }}{{ end -}} {{- /* The middle block (category, title, summary) is centered in the band between the header and the rule, y 120 to 500. og-wrap.html breaks the text into lines that fit the column, which gives the block's height: a line is the font's ascent plus the line gap. The title takes up to 5 lines; the summary gets what is left, at most 3, cut with "...". */ -}} {{- $column := 560 -}} {{- $band := 380 -}} {{- $ascent := hugo.Data.ogcard.Bold.ascent -}} {{- $eyebrowStep := cond (ne $eyebrow "") (add (math.Ceil (mul $ascent 22)) 16) 0 -}} {{- $titleStep := add (math.Ceil (mul $ascent 50)) 12 -}} {{- $summaryStep := add (math.Ceil (mul hugo.Data.ogcard.Medium.ascent 24)) 10 -}} {{- $title := partial "og-wrap.html" (dict "text" $heading "font" "Bold" "size" 50 "width" $column "max" 5) -}} {{- $used := add $eyebrowStep (sub (mul (len $title) $titleStep) 12) -}} {{- $lines := slice -}} {{- with $summary -}} {{- $room := int (math.Floor (div (sub $band $used 22 8 -10) $summaryStep)) -}} {{- if gt $room 3 }}{{ $room = 3 }}{{ end -}} {{- if gt $room 0 -}} {{- $lines = partial "og-wrap.html" (dict "text" . "font" "Medium" "size" 24 "width" $column "max" $room) -}} {{- end -}} {{- end -}} {{- $total := add $used 8 -}} {{- with $lines }}{{ $total = add $total 22 (sub (mul (len .) $summaryStep) 10) }}{{ end -}} {{- $y := int (add 120 (div (sub $band $total) 2)) -}} {{- $filters := slice (images.Text site.Title (dict "font" $fonts.Bold "size" 32 "color" $ink "x" 413 "y" 44)) (images.Text $footer (dict "font" $fonts.Medium "size" 22 "color" $muted "x" 600 "y" 550 "alignx" "center")) -}} {{- with $eyebrow -}} {{- $filters = $filters | append (images.Text . (dict "font" $fonts.Bold "size" 22 "color" $accent "x" 600 "y" $y "alignx" "center")) -}} {{- $y = add $y $eyebrowStep -}} {{- end -}} {{- range $title -}} {{- $filters = $filters | append (images.Text . (dict "font" $fonts.Bold "size" 50 "color" $ink "x" 600 "y" $y "alignx" "center")) -}} {{- $y = add $y $titleStep -}} {{- end -}} {{- $y = add $y -12 22 -}} {{- range $lines -}} {{- $filters = $filters | append (images.Text . (dict "font" $fonts.Medium "size" 24 "color" $muted "x" 600 "y" $y "alignx" "center")) -}} {{- $y = add $y $summaryStep -}} {{- end -}} {{- $card := resources.Get "og/card-base.png" | images.Filter $filters -}} {{- return $card -}}
