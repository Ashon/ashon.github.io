{{- /* og-wrap.html - breaks .text into lines no wider than .width pixels, at spaces, for the share card (og-card.html), which then draws each line centered on its own. Returns the lines as a slice. With .max, the text is cut to that many lines and the last one ends in "...". Glyph widths come from data/ogcard.json (scripts/og-glyph-widths.py), rounded up, so a line never comes out wider than measured. A single word wider than .width keeps a line to itself. Params: text, font (Bold or Medium), size, width, max (optional). */ -}} {{- $font := index hugo.Data.ogcard .font -}} {{- $size := .size -}} {{- $edge := .width -}} {{- $max := .max | default 0 -}} {{- $space := mul (index $font.chars " ") $size -}} {{- $ellipsis := mul (index $font.chars ".") 3 $size -}} {{- $rows := slice -}} {{- $row := slice -}} {{- $widths := slice -}} {{- $x := 0.0 -}} {{- $cut := false -}} {{- range split .text " " -}} {{- if and . (not $cut) -}} {{- $w := 0.0 -}} {{- range split . "" -}} {{- $em := index $font.chars . -}} {{- if not $em }}{{ $em = cond (findRE `^[가-힣]$` .) $font.hangul 1.0 }}{{ end -}} {{- $w = add $w (mul $em $size) -}} {{- end -}} {{- if and $row (gt (add $x $w) $edge) -}} {{- if and $max (ge (add (len $rows) 1) $max) -}} {{- $cut = true -}} {{- else -}} {{- $rows = $rows | append (delimit $row " ") -}} {{- $row = slice -}} {{- $widths = slice -}} {{- $x = 0.0 -}} {{- end -}} {{- end -}} {{- if not $cut -}} {{- $row = $row | append . -}} {{- $widths = $widths | append (add $w $space) -}} {{- $x = add $x $w $space -}} {{- end -}} {{- end -}} {{- end -}} {{- if $cut -}} {{- /* "..." joins the last word, so drop words off the end of the last line until that word plus "..." fits. */ -}} {{- range seq 20 -}} {{- if and (gt (add (sub $x $space) $ellipsis) $edge) (gt (len $row) 1) -}} {{- $x = sub $x (index $widths (sub (len $widths) 1)) -}} {{- $widths = first (sub (len $widths) 1) $widths -}} {{- $row = first (sub (len $row) 1) $row -}} {{- end -}} {{- end -}} {{- end -}} {{- with $row }}{{ $rows = $rows | append (printf "%s%s" (delimit . " ") (cond $cut "..." "")) }}{{ end -}} {{- return $rows -}}
